def synergistic_stereo(nL, nF, sVal, zFoc, dLux, sVal_ref, uNrm_ref, zSng_ref):
# The synergistic stereo main algorithm.
# 
# Input in a "production" run is an array of blurry images {sVal[L][F]}
# where {L} (the light soruce index) varies in {0..nL-1} and {F} (the
# focusing index) varies in {0..nF-1}. Each light source {L} is a
# point-like at infinity in the direction {dLux[L]} . Also input is the
# {Z} coordinate {zFoc[F]} of the in-focus plane for each focusing index
# {F}.
#
# Output in a "production" run is two images {zSng} and {wSng}.
# The value of {zSng[x,y]} is the estimated average surface height ({Z} coordinate)
# in the pixel {[x,y]}.  The value {wSng[x,y]} is the reliability 
# or confidence weight of that estimate.
#
# If the images are generated synthetically for testing purpose, there
# will be also a sharp image {sVal_ref[L]} for each light {L}, a single
# reference normal map {uNrm_ref}, and a single reference height map
# {zSng_ref}, and. These may be used to verify the results of the
# algorithm, but are not used by it.
#

    # PART 1. Running the multifocus stereo.

    # Step 1.1. Averaging the light sources for each focusing depth.

    for F in range(nF):
      sVal_med[F] = average((sVal[L][F] for L in range(nL)))
    sVal_ref_med = average((sVal_ref[L] for L in range(nL)))

    # Step 1.2. Applying the multifocus algorithm proper.

    iSel, wSel = multifocus_stereo(nF, sVal_med)

    # Step 1.3. Applying the mosaicking algorithm.

    for L in range(nL):
      sMos[L], zMos[L] = mosaic(nF, iSel, (sVal[L][F] for F in range(nF)), zFoc, sVal_ref[L], zSng_ref):
    sMos_med, zMos_med = mosaic(nF, iSel, sVal_med, zFoc, sVal_ref_med, zSng_ref):
    # Note: {zMos[L]} and {zMos_med} should be identical for all {L}.

    # PART 2. Running the photometric stereo: 

    uNrm, wNrm = photometric_stereo(nL, sMos, dLux, uNrm_ref);

    # PART 3. Synergistic integration:

    zSng, wSng = integrate(zMos_med, wSel, uNrm, wNrm, zSng_ref);

    return zSng, wSng
